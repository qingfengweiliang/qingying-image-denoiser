cmake_minimum_required(VERSION 3.22)
project(qingying_core_tests)
find_package(ncnn REQUIRED)
find_package(OpenMP REQUIRED)
set(SRC ${CMAKE_CURRENT_SOURCE_DIR}/../../app/src/main/cpp)
add_executable(smoke smoke.cpp ${SRC}/waifu2x.cpp ${SRC}/realcugan.cpp)
target_include_directories(smoke PRIVATE ${SRC})
target_compile_definitions(smoke PRIVATE ASSETS_DIR="${CMAKE_CURRENT_SOURCE_DIR}/../../app/src/main/assets/")
target_link_libraries(smoke ncnn OpenMP::OpenMP_CXX)
enable_testing()
add_test(NAME models COMMAND smoke)
